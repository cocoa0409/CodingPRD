# Example Test Patterns in Python

Concrete pytest code patterns organized to match the three-layer architecture described in
SKILL.md: **`prd_bench_runner.testing` (framework)** → **`test_helpers/` (PRD-specific)** → **`test_suite/` (tests)**.

Generic utilities (HTTP client base, validators, format checkers, race helpers) live in
`prd_bench_runner.testing` and are **NOT generated** — they are imported.
Only PRD-specific code (entity types, endpoint methods, payload factories, domain assertions,
fixtures) is generated into `test_helpers/`.

All patterns use **domain-agnostic placeholder names** (`{Entity}`, `{parent}`, etc.).
Replace with the PRD's actual domain terminology when generating.

```python
"""Entity assertions and domain invariant checkers."""
import pytest
from prd_bench_runner.testing import (
    is_valid_uuid, is_iso8601_timestamp, assert_cost_precision,
)

def assert_{entity}(data: dict):
    """Validate {Entity} response structure against PRD contract."""
    assert is_valid_uuid(data.get("id", "")), "id must be valid UUID"
    assert is_iso8601_timestamp(data.get("created_at", "")), "created_at must be ISO 8601"
    assert is_iso8601_timestamp(data.get("updated_at", "")), "updated_at must be ISO 8601"
    assert isinstance(data["field_a"], str) and len(data["field_a"]) > 0
    assert data["status"] in ["STATE_A", "STATE_B", "STATE_C"]
    assert isinstance(data["computed_field"], (int, float))
    assert data["computed_field"] >= 0
    assert isinstance(data.get("version"), int)

def assert_{parent}_invariants(data: dict):
    """Domain invariants that must hold after ANY operation."""
```

## 1.8 `fixtures.py`

The most substantial file in `test_helpers/`. It encapsulates multi-step setup sequences so
tests focus on assertions, not plumbing. This file is expected to be large (500–1000+ lines)
because most interesting test claims require 5–15 API calls to reach the testable state.

```python
"""
Composite setup helpers for multi-step test prerequisites.
Uses get_success_data() from framework (lenient validator) so setup failures
produce clear "setup failed" errors, not confusing assertion failures.
"""
from prd_bench_runner.testing import get_success_data
from .api_client import api_client
from .factories import *
from .constants import THRESHOLDS, TEST_DEFAULTS
from .test_users import TEST_USERS

# ─── Tier 1: State Progression ────────────────────────
# For entities with state machines where testable state ≠ initial state.

def create_and_activate_{parent}(budget=250_000.00, name="Test Plan"):
    """Create {Parent} and transition to ACTIVE. Returns activated entity."""
    res = api_client.create_{parent}(
        create_{parent}_payload(name=name, budget=budget),
        TEST_USERS["role_a_headers"]
    )
    entity = get_success_data(res)
    activate_res = api_client.activate_{parent}(entity["id"], TEST_USERS["role_a_headers"])
    return get_success_data(activate_res)


# ─── Tier 2: Entity Hierarchy ─────────────────────────
# For PRDs with parent-child relationships.

def setup_{parent}_with_{child}_and_{resource}(**overrides):
    """Active parent + one child + one resource. Returns (parent, child, resource)."""
    parent = create_and_activate_{parent}(budget=overrides.get("budget", 250_000.00))

    child_res = api_client.create_{child}(parent["id"],
                create_{child}_payload(**overrides.get("child_overrides", {})),
                TEST_USERS["role_b_headers"])
    child = get_success_data(child_res)

    resource_res = api_client.create_{resource}(parent["id"],
                    create_{resource}_payload(**overrides.get("resource_overrides", {})),
                    TEST_USERS["role_b_headers"])
    resource = get_success_data(resource_res)
    return parent, child, resource

def setup_basic_scenario():
    """Active parent + child + resource + one procedure. Most common setup."""
    parent, child, resource = setup_{parent}_with_{child}_and_{resource}()
    proc = get_success_data(api_client.create_procedure(
        parent["id"], child["id"], create_procedure_payload(),
        TEST_USERS["role_b_headers"]))
    return parent, child, resource, proc


# ─── Tier 3: Full Lifecycle Shortcuts ─────────────────
# For multi-step lifecycles (create → approve → schedule).

def create_and_schedule_assignment(parent_id, procedure_id, resource_id):
    """Create → approve → schedule. Returns scheduled assignment."""
    asg = get_success_data(api_client.create_assignment(
        parent_id, create_assignment_payload(procedure_id, resource_id),
        TEST_USERS["role_b_headers"]))
    api_client.protocol_approve(parent_id, asg["id"], TEST_USERS["role_a_headers"])
    sched_res = api_client.schedule(parent_id, asg["id"], TEST_USERS["role_a_headers"])
    return get_success_data(sched_res)

def create_scheduled_assignment(parent_id, child_id, resource_id,
                                 proc_overrides=None, asg_overrides=None):
    """Create procedure + assignment, walk through to SCHEDULED. Returns (proc, asg)."""
    proc = get_success_data(api_client.create_procedure(
        parent_id, child_id,
        create_procedure_payload(**(proc_overrides or {})),
        TEST_USERS["role_b_headers"]))
    asg = get_success_data(api_client.create_assignment(
        parent_id, create_assignment_payload(proc["id"], resource_id, **(asg_overrides or {})),
        TEST_USERS["role_b_headers"]))
    api_client.protocol_approve(parent_id, asg["id"], TEST_USERS["role_a_headers"])
    sched_res = api_client.schedule(parent_id, asg["id"], TEST_USERS["role_a_headers"])
    return proc, get_success_data(sched_res)

def create_multiple_scheduled(parent_id, child_id, resource_id, count,
                               start_minute=100, duration=60, spacing=100):
    """Create N scheduled assignments in sequence. Returns list of (proc, asg)."""
    results = []
    current = start_minute
    for _ in range(count):
        proc, asg = create_scheduled_assignment(
            parent_id, child_id, resource_id,
            proc_overrides={"start_minute": current, "end_minute": current + duration}
        )
        results.append((proc, asg))
        current += duration + spacing
    return results


# ─── Tier 4: Accumulation / Threshold Targeting ───────
# CRITICAL for boundary-precision testing. Without these, tests cannot
# reach the exact cumulative state needed for threshold assertions.

def setup_with_cumulative_usage(minutes_used: int, cost_per_minute: float = 1.0):
    """
    Create a plan where resource already has `minutes_used` of SCHEDULED time.
    For boundary tests: set minutes_used = THRESHOLD - 20, then test at boundary.

    Internally: plan → activate → rotation → resource → one large procedure
    spanning `minutes_used` minutes → assign → approve → schedule.
    """
    parent = create_and_activate_{parent}(budget=1_000_000)
    child = get_success_data(api_client.create_{child}(
        parent["id"], create_{child}_payload(), TEST_USERS["role_b_headers"]))
    resource = get_success_data(api_client.create_{resource}(
        parent["id"],
        create_{resource}_payload(operating_cost_per_minute=cost_per_minute),
        TEST_USERS["role_b_headers"]))

    if minutes_used > 0:
        proc = get_success_data(api_client.create_procedure(
            parent["id"], child["id"],
            create_procedure_payload(start_minute=0, end_minute=minutes_used),
            TEST_USERS["role_b_headers"]))
        asg = get_success_data(api_client.create_assignment(
            parent["id"], create_assignment_payload(proc["id"], resource["id"]),
            TEST_USERS["role_b_headers"]))
        api_client.protocol_approve(parent["id"], asg["id"], TEST_USERS["role_a_headers"])
        api_client.schedule(parent["id"], asg["id"], TEST_USERS["role_a_headers"])

    return parent, child, resource

def setup_and_raise_score(target_score, plan_budget=200_000, resource_cost=2.0):
    """
    Accumulate safety/risk score to `target_score` by creating N penalty-triggering
    assignments. Uses domain constants to calculate how many are needed.

    Internally:
    1. Create plan + activate + rotation + TWO resources (main + penalty)
    2. Schedule one large base procedure (e.g., 5950 min) to enter penalty zone
    3. Loop: create penalty procedure → assign → approve → schedule
       Each adds PENALTY_AMOUNT to score. If score >= HARD_LIMIT, uses override.
    """
    parent, child = create_and_activate_{parent}(budget=plan_budget), ...
    # ... create main_resource and penalty_resource ...

    penalty = THRESHOLDS["HIGH_BIORISK_PENALTY"]       # e.g., 50
    override_penalty = THRESHOLDS["OVERRIDE_PENALTY"]   # e.g., 100
    hard_limit = THRESHOLDS["SAFETY_SCORE_HARD_LIMIT"]  # e.g., 500

    # Base procedure to enter penalty zone
    # ... create and schedule base procedure of ~5950 minutes ...

    current_score = 0
    needed = 0
    while current_score < target_score:
        needs_override = (current_score + penalty) >= hard_limit
        # Create penalty procedure → assign (with override if needed) → approve → schedule
        if needs_override:
            current_score += penalty + override_penalty
        else:
            current_score += penalty
        needed += 1

    return parent, main_resource, child
```

### Why threshold/accumulation fixtures are so important

Consider testing "bio-risk triggered when cumulative minutes exceed 6000":

**Without fixture (wrong — test is trivial):**
```python
def test_biorisk_threshold(api, headers):
    plan = create_and_activate_plan(api, headers)
    # Creates ONE assignment and checks is_high_biorisk
    # This CANNOT test the boundary — cumulative is only 60 minutes
```

**With fixture (correct — reaches exact boundary):**
```python
def test_biorisk_at_6001_minutes(api, headers):
    parent, child, resource = setup_with_cumulative_usage(minutes_used=5980)
    # System already has 5980 SCHEDULED minutes on this resource.
    # Now test: 21-minute procedure → cumulative = 6001 → bio-risk triggered
    proc = create_procedure(parent, child, minutes=21)
    asg = create_assignment(parent, proc, resource)
    assert asg["is_high_biorisk"] is True
```

The fixture does 7 API calls (plan → activate → rotation → resource → procedure(5980 min) →
assignment → approve → schedule) so the test can focus on its single 3-line assertion.


# ─── Tier 5: Finalization / Terminal State ────────────

```python
def setup_finalization_ready(num_assignments=3, budget=10_000, cost_per_minute=2):
    """
    Full setup through FINAL_REVIEW_PENDING:
    plan → activate → rotation → resource → N scheduled assignments →
    transition child_container to READY → initiate final review.
    Returns (plan, rotation, resource, assignments, procedures).
    """
    parent, child, resource = setup_{parent}_with_{child}_and_{resource}(
        budget=budget, resource_overrides={"operating_cost_per_minute": cost_per_minute})
    results = create_multiple_scheduled(
        parent["id"], child["id"], resource["id"], num_assignments)
    # Transition child container to terminal-ready state
    api_client.update_status(parent["id"], child["id"],
        {"status": "READY_FOR_ASSIGNMENT"}, TEST_USERS["role_b_headers"])
    api_client.initiate_final_review(parent["id"], TEST_USERS["role_a_headers"])
    return parent, child, resource, [r[1] for r in results], [r[0] for r in results]

def setup_finalized_plan(budget=1_000, cost_per_minute=20):
    """Entire lifecycle through COMPLETED. For post-finalization lockdown tests."""
    parent, child, resource, assignments, _ = setup_finalization_ready(
        num_assignments=1, budget=budget, cost_per_minute=cost_per_minute)
    api_client.finalize(parent["id"], TEST_USERS["admin_headers"])
    final_plan = get_success_data(
        api_client.get_{parent}(parent["id"], TEST_USERS["role_a_headers"]))
    return final_plan, assignments[0]
```

## 2.1 CRUD & Input Validation

```python
def test_create_and_retrieve_{entity}(api, role_a_headers):
    """Table 1 claim: POST creates entity; GET retrieves it with all fields."""
    payload = create_{entity}_payload(name="Test Entity Alpha")
    res = api.create_{entity}(payload, role_a_headers)
    entity = validate_success(res, 201)
    assert_{entity}(entity)
    assert entity["name"] == "Test Entity Alpha"
    assert entity["status"] == "INITIAL_STATE"

    get_res = api.get_{entity}(entity["id"], role_a_headers)
    retrieved = validate_success(get_res, 200)
    assert retrieved["id"] == entity["id"]

def test_list_with_pagination(api, role_a_headers):
    """Table 5 claim: List endpoint supports limit/offset with correct meta."""
    for i in range(3):
        api.create_{entity}(create_{entity}_payload(name=f"Entity {i}"), role_a_headers)

    data, meta = validate_paginated(api.list_{entities}(role_a_headers))
    assert meta["total_count"] >= 3

    data, meta = validate_paginated(api.list_{entities}(role_a_headers, params={"limit": 1}))
    assert len(data) == 1
    assert meta["has_more"] is True

    data, meta = validate_paginated(api.list_{entities}(role_a_headers, params={"offset": 9999}))
    assert len(data) == 0
    assert meta["has_more"] is False

def test_get_nonexistent_{entity}(api, role_a_headers):

@pytest.mark.parametrize("missing_field", ["field_a", "field_b", "field_c"])

def test_cannot_create_child_in_wrong_parent_state(api, role_b_headers):
```

## 2.2 State Machines

```python
def test_valid_transition_{from}_to_{to}(api, role_a_headers):
    """Table 2 claim: {trigger} transitions from {FROM} to {TO}."""
    entity = create_and_activate_{parent}()
    old_version = entity["version"]
    res = api.{next_trigger}(entity["id"], role_a_headers)
    updated = validate_success(res, 200)
    assert updated["status"] == "TO_STATE"
    assert_version_incremented(updated["version"], old_version)

@pytest.mark.parametrize("current_state,trigger,expected_error", [
    ("ACTIVE", "activate", "ERR_{ENTITY}_NOT_IN_PLANNING"),
    ("COMPLETED", "cancel", "ERR_{ENTITY}_NOT_CANCELLABLE"),
])
def test_invalid_state_transition(api, role_a_headers, current_state, trigger, expected_error):
    """Table 2 claim: Invalid trigger from wrong state → correct error."""
    # ... setup entity in current_state
```
